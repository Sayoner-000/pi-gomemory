// Tests 01-04 assert safe behavior and fail against the frozen target.
// Test 05 is a bounded witness, not an exhaustion test or a threshold contract.
func TestACRADM01NewPinOverridesOldCache(t *testing.T) {
 p := filepath.Join(t.TempDir(), "cache.json")
 b, _ := json.Marshal(depsCache{CheckedAt: time.Now(), Deps: []DepState{{Name:"pi-antigravity", Current:"0.12.0", Latest:"0.12.0"}}})
 if err := os.WriteFile(p,b,0600); err != nil { t.Fatal(err) }
 probes := []DepProbe{{Name:"pi-antigravity", Current:fixedVersion("0.12.0"), Latest:fixedVersion(AntigravityVersion), Update:[]string{"pi","install",AntigravityPackageSource}}}
 got := DepsStatusFor(p, probes)
 if len(got) != 1 || got[0].Latest != AntigravityVersion { t.Fatalf("old cache hides new approved pin %s: got %+v",AntigravityVersion,got) }
}

func TestACRADM02PreserveSymlinkAfterPlan(t *testing.T) {
 d := t.TempDir()
 defaults := []byte(`{"showStartupHeader":false}`)
 cfg := filepath.Join(d,"pi-cc-extensions.json")
 target := filepath.Join(d,"personal.json")
 if err := os.WriteFile(cfg, defaults,0600); err != nil {t.Fatal(err)}
 if err := os.WriteFile(target, defaults,0600); err != nil {t.Fatal(err)}
 p := &UninstallPlan{}
 p.planCCExtensions(fstest.MapFS{"config/pi-cc-extensions.json": &fstest.MapFile{Data:defaults}},d)
 if len(p.Actions) != 1 || p.Actions[0].apply == nil {t.Fatalf("no removal planned: %+v",p.Actions)}
 if err := os.Remove(cfg); err != nil {t.Fatal(err)}
 if err := os.Symlink(target,cfg); err != nil {t.Fatal(err)}
 err := p.Actions[0].apply()
 if _, statErr := os.Lstat(cfg); statErr != nil {t.Fatalf("new personal symlink removed: apply=%v lstat=%v",err,statErr)}
}

func TestACRADM05EmptyEntriesAreBounded(t *testing.T) {
 // A bounded sample demonstrates that payload accounting stays at zero,
 // not an attempt to exhaust resources on this machine.
 var b bytes.Buffer
 gz := gzip.NewWriter(&b)
 tw := tar.NewWriter(gz)
 const count = 2000
 for i:=0; i<count; i++ {
  name := filepath.ToSlash(filepath.Join("package",time.Unix(int64(i),0).UTC().Format("150405")))
  if err:=tw.WriteHeader(&tar.Header{Name:name,Mode:0600,Size:0,Typeflag:tar.TypeReg}); err!=nil {t.Fatal(err)}
 }
 if err:=tw.Close();err!=nil {t.Fatal(err)}
 if err:=gz.Close();err!=nil {t.Fatal(err)}
 d:=t.TempDir()
 err:=extractCandidate(b.Bytes(),d)
 n:=0
 _=filepath.WalkDir(d,func(_ string,e os.DirEntry,err error)error{if err==nil && !e.IsDir(){n++};return err})
 t.Logf("compressed=%d bytes, payload=0, extracted files=%d, extraction error=%v",b.Len(),n,err)
 if err!=nil || n!=count {t.Fatalf("bounded witness did not reproduce: entries=%d err=%v",n,err)}
}
